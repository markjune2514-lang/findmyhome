        function setCookie(cname, cvalue, exdays) {
            const d = new Date();
            d.setTime(d.getTime() + (exdays * 24 * 60 * 60 * 1000));
            let expires = "expires=" + d.toUTCString();
            document.cookie = cname + "=" + cvalue + ";" + expires + ";path=/";
        }

        function getCookie(cname) {
            let name = cname + "=";
            let decodedCookie = decodeURIComponent(document.cookie);
            let ca = decodedCookie.split(';');
            for (let i = 0; i < ca.length; i++) {
                let c = ca[i];
                while (c.charAt(0) == ' ') {
                    c = c.substring(1);
                }
                if (c.indexOf(name) == 0) {
                    return c.substring(name.length, c.length);
                }
            }
            return "";
        }

(function () {
    'use strict';

    /* — Constants — */
    var ANIMATION_DURATION_MS = 600;
    var ANIMATION_EASING      = 'cubic-bezier(0.4, 0, 0.2, 1)';
    var BG_CROSSFADE_DELAY_MS = 850;
    var RESIZE_DEBOUNCE_MS    = 150;
    var MOBILE_BREAKPOINT     = 991;
    var TABLET_MIN_WIDTH      = 768;
    var DESKTOP_THUMB_WIDTH   = 220;
    var MOBILE_PEEK_WIDTH     = 40;
    var ACTIVE_SLIDE_RATIO    = 0.45;

    /* — DOM References — */
    var track       = document.getElementById('facilityTrack');
    var box         = document.getElementById('facilityBox');
    var bgBack      = document.getElementById('facilityBgBack');
    var bgFront     = document.getElementById('facilityBgFront');
    var heroImg     = document.getElementById('facilityHeroImg');
    var mobileTitle = document.getElementById('facilityMobileTitle');
    var mobileDesc  = document.getElementById('facilityMobileDesc');
    var btnPrev     = document.getElementById('facilityBtnPrev');
    var btnNext     = document.getElementById('facilityBtnNext');

    var isAnimating = false;

    /* — Breakpoint Helpers — */

    /** @returns {boolean} True when viewport is at or below mobile breakpoint */
    function isMobileViewport() {
        return window.innerWidth <= MOBILE_BREAKPOINT;
    }

    /** @returns {boolean} True when viewport is in the tablet range */
    function isTabletViewport() {
        return window.innerWidth >= TABLET_MIN_WIDTH && isMobileViewport();
    }

    /**
     * Returns responsive layout configuration.
     * @returns {{ gap: number, columns: number|null, padding: number }}
     */
    function getResponsiveConfig() {
        if (isTabletViewport()) return { gap: 12, columns: 3, padding: 32 };
        if (isMobileViewport()) return { gap: 10, columns: 2, padding: 24 };
        return { gap: 15, columns: null, padding: 40 };
    }

    /* — Slide Queries & Sizing — */

    /** @returns {HTMLElement[]} All slide elements in current DOM order */
    function getAllSlides() {
        return [].slice.call(track.querySelectorAll('.facility-slide'));
    }

    /**
     * Calculates the width of a single thumbnail image.
     * Desktop: fixed width. Mobile/Tablet: fills available columns with peek space.
     * @returns {number}
     */
    function calculateThumbWidth() {
        var cfg = getResponsiveConfig();
        if (!cfg.columns) return DESKTOP_THUMB_WIDTH;

        var availableWidth = box.clientWidth - cfg.padding - cfg.gap * (cfg.columns - 1) - MOBILE_PEEK_WIDTH;
        return Math.floor(availableWidth / cfg.columns);
    }

    /**
     * Calculates the width of the active (expanded) slide.
     * Mobile: same as thumb. Desktop: percentage of box inner width.
     * @returns {number}
     */
    function calculateActiveSlideWidth() {
        if (isMobileViewport()) return calculateThumbWidth();

        var cfg = getResponsiveConfig();
        return Math.floor((box.clientWidth - cfg.padding) * ACTIVE_SLIDE_RATIO);
    }

    /* — Mobile Active Info Sync — */

    /**
     * Copies the active slide's title and description into the mobile info panel.
     * @param {HTMLElement} slide - The currently active slide element
     */
    function syncMobileActiveInfo(slide) {
        var infoPanel = slide.querySelector('.facility-slide-info');
        if (!infoPanel) return;

        mobileTitle.textContent = infoPanel.querySelector('.slide-title').textContent;
        mobileDesc.textContent  = infoPanel.querySelector('.slide-description').textContent;
    }

    /* — Background Crossfade — */

    /**
     * Crossfades the full-bleed background image to a new source.
     * Skips if the image is already showing.
     * @param {string} imageUrl - URL of the new background image
     */
    function crossfadeBackground(imageUrl) {
        var cssValue = 'url(' + imageUrl + ')';
        if (bgBack.style.backgroundImage === cssValue) return;

        bgFront.style.backgroundImage = cssValue;
        void bgFront.offsetHeight; /* force reflow for transition */
        bgFront.classList.add('is-active');

        setTimeout(function () {
            bgBack.style.backgroundImage = cssValue;
            bgFront.classList.remove('is-active');
        }, BG_CROSSFADE_DELAY_MS);
    }

    /* — Layout Engine — */

    /**
     * Recalculates and applies widths to all slides.
     * On mobile: positions nav buttons at the vertical center of the track.
     * On desktop: resets any mobile-specific inline styles.
     */
    function applyLayout(skipBackground) {
        var slides = getAllSlides();
        if (!slides.length) return;

        /* ── Read phase (avoid layout thrashing) ── */
        var cfg          = getResponsiveConfig();
        var thumbWidth   = calculateThumbWidth();
        var activeWidth  = calculateActiveSlideWidth();
        var mobile       = isMobileViewport();
        var infoWidth    = mobile ? 0 : activeWidth - thumbWidth - cfg.gap;

        /* ── Write phase ── */
        slides.forEach(function (slide, index) {
            var isActiveSlide = index === 0;
            var infoPanel     = slide.querySelector('.facility-slide-info');
            var infoInner     = slide.querySelector('.slide-info-inner');

            slide.classList.toggle('is-active', isActiveSlide);

            /* Always set inner width so text is pre-laid-out at full width */
            infoInner.style.width = infoWidth + 'px';

            if (mobile || !isActiveSlide) {
                slide.style.width       = thumbWidth + 'px';
                infoPanel.style.width      = '0';
                infoPanel.style.marginLeft = '0';
            } else {
                slide.style.width          = activeWidth + 'px';
                infoPanel.style.width      = infoWidth + 'px';
                infoPanel.style.marginLeft = cfg.gap + 'px';
            }
        });

        /* ── Update active content ── */
        var activeImageUrl = slides[0].getAttribute('data-img');

        if (mobile) {
            syncMobileActiveInfo(slides[0]);
            heroImg.style.backgroundImage = 'url(' + activeImageUrl + ')';
            positionNavButtonsAtTrackCenter();
        } else {
            resetNavButtonPositions();
        }

        if (!skipBackground) crossfadeBackground(activeImageUrl);
    }

    /**
     * On mobile, positions nav buttons vertically at the center of the track area,
     * accounting for the box's offset and padding.
     */
    function positionNavButtonsAtTrackCenter() {
        /* box.offsetTop = distance from .facility-slider to .facility-box
           track.offsetTop = distance from .facility-box to .facility-track (includes mobile-info height + padding) */
        var buttonTop = (box.offsetTop + track.offsetTop + track.offsetHeight / 2) + 'px';

        btnPrev.style.top = buttonTop;
        btnNext.style.top = buttonTop;
    }

    /** Clears mobile-specific inline positioning from nav buttons. */
    function resetNavButtonPositions() {
        btnPrev.style.top = '';
        btnNext.style.top = '';
    }

    /* — Track Animation — */

    /**
     * Animates the track element from one X position to another.
     * Uses CSS transitions with a safety timeout fallback.
     * @param {number} fromX  - Starting translateX value (px)
     * @param {number} toX    - Ending translateX value (px)
     * @param {Function} onComplete - Called once after animation settles
     */
    function animateTrack(fromX, toX, onComplete) {
        track.style.transition = 'none';
        track.style.transform  = 'translateX(' + fromX + 'px)';
        void track.offsetHeight; /* force reflow */

        track.style.transition = 'transform ' + ANIMATION_DURATION_MS + 'ms ' + ANIMATION_EASING;
        track.style.transform  = 'translateX(' + toX + 'px)';

        var hasSettled = false;

        function settle() {
            if (hasSettled) return;
            hasSettled = true;
            track.removeEventListener('transitionend', onTransitionEnd);
            track.style.transition = 'none';
            track.style.transform  = 'translateX(0)';
            isAnimating = false;
            onComplete();
        }

        function onTransitionEnd(event) {
            if (event.target === track) settle();
        }

        track.addEventListener('transitionend', onTransitionEnd);
        setTimeout(settle, ANIMATION_DURATION_MS + 100); /* safety fallback */
    }

    /* — Navigation — */

    /**
     * Navigates the carousel by one slide in the given direction.
     * @param {number} direction - 1 for next, -1 for previous
     */
    function navigateByOne(direction) {
        if (isAnimating) return;

        var slides = getAllSlides();
        if (slides.length <= 1) return;

        isAnimating = true;
        var gap = getResponsiveConfig().gap;

        if (direction === 1) {
            var slideWidth = slides[0].offsetWidth + gap;
            animateTrack(0, -slideWidth, function () {
                track.appendChild(slides[0]);
                applyLayout();
            });
        } else {
            var lastSlide = slides[slides.length - 1];
            track.insertBefore(lastSlide, slides[0]);
            applyLayout(true);

            var newFirstWidth = getAllSlides()[0].offsetWidth + gap;
            animateTrack(-newFirstWidth, 0, function () {
                applyLayout();
            });
        }
    }

    /**
     * Navigates directly to a slide at the given index.
     * @param {number} targetIndex - Zero-based index of the target slide
     */
    function navigateToSlide(targetIndex) {
        if (isAnimating || targetIndex === 0) return;

        var slides = getAllSlides();
        if (targetIndex >= slides.length) return;

        isAnimating = true;
        var gap = getResponsiveConfig().gap;

        var totalDistance = 0;
        for (var i = 0; i < targetIndex; i++) {
            totalDistance += slides[i].offsetWidth + gap;
        }

        animateTrack(0, -totalDistance, function () {
            for (var i = 0; i < targetIndex; i++) {
                track.appendChild(getAllSlides()[0]);
            }
            applyLayout();
        });
    }

    /* — Event Binding — */

    btnNext.addEventListener('click', function () { navigateByOne(1); });
    btnPrev.addEventListener('click', function () { navigateByOne(-1); });

    track.addEventListener('click', function (event) {
        var clickedSlide = event.target.closest('.facility-slide');
        if (clickedSlide) {
            navigateToSlide(getAllSlides().indexOf(clickedSlide));
        }
    });

    /* — Drag / Swipe — */
    var DRAG_THRESHOLD = 30;
    var dragStartX     = 0;
    var dragCurrentX   = 0;
    var isDragging     = false;
    var hasDragged     = false;

    function onDragStart(x) {
        if (isAnimating) return;
        isDragging   = true;
        hasDragged   = false;
        dragStartX   = x;
        dragCurrentX = x;
        track.style.transition = 'none';
    }

    function onDragMove(x) {
        if (!isDragging) return;
        dragCurrentX = x;
        var delta = dragCurrentX - dragStartX;
        if (Math.abs(delta) > 5) hasDragged = true;
        track.style.transform = 'translateX(' + delta + 'px)';
    }

    function onDragEnd() {
        if (!isDragging) return;
        isDragging = false;
        var delta = dragCurrentX - dragStartX;

        /* Reset transform immediately */
        track.style.transition = 'none';
        track.style.transform  = 'translateX(0)';

        if (Math.abs(delta) >= DRAG_THRESHOLD) {
            navigateByOne(delta < 0 ? 1 : -1);
        }
    }

    /* Mouse events */
    track.addEventListener('mousedown', function (e) {
        e.preventDefault();
        onDragStart(e.clientX);
    });
    document.addEventListener('mousemove', function (e) {
        onDragMove(e.clientX);
    });
    document.addEventListener('mouseup', function () {
        onDragEnd();
    });

    /* Touch events */
    track.addEventListener('touchstart', function (e) {
        onDragStart(e.touches[0].clientX);
    }, { passive: true });
    track.addEventListener('touchmove', function (e) {
        if (isDragging && hasDragged) e.preventDefault();
        onDragMove(e.touches[0].clientX);
    }, { passive: false });
    track.addEventListener('touchend', function () {
        onDragEnd();
    });

    /* Prevent click when dragged */
    track.addEventListener('click', function (e) {
        if (hasDragged) {
            e.stopImmediatePropagation();
            e.preventDefault();
        }
    }, true);

    var resizeTimer;
    window.addEventListener('resize', function () {
        clearTimeout(resizeTimer);
        resizeTimer = setTimeout(applyLayout, RESIZE_DEBOUNCE_MS);
    });

    /* — Bootstrap — */

    /**
     * Initializes the carousel: sets the first background image,
     * runs layout twice (once to measure, once after paint to stabilize),
     * then reveals the track.
     */
    function initialize() {
        var slides = getAllSlides();
        if (!slides.length) return;

        /* ── Build SEO alt/title from dataLayer ── */
        var dlData = {};
        if (window.dataLayer) {
            for (var i = 0; i < window.dataLayer.length; i++) {
                var entry = window.dataLayer[i];
                if (entry.projectNameTH) { dlData = entry; break; }
            }
        }

        if (dlData.propertyTypeTH && dlData.projectNameTH && dlData.projectNameEN) {
            var prefix = dlData.propertyTypeTH + ' ' + dlData.projectNameTH + ' (' + dlData.projectNameEN + ') พื้นที่ส่วนกลาง Facility - ';
            slides.forEach(function (slide) {
                var facilityTitle = slide.getAttribute('data-facility-title') || '';
                var altText = prefix + facilityTitle;
                var img = slide.querySelector('img');
                if (img) {
                    img.alt   = altText;
                    img.title = altText;
                }
            });
        }

        var firstImageUrl = 'url(' + slides[0].getAttribute('data-img') + ')';
        bgBack.style.backgroundImage  = firstImageUrl;
        heroImg.style.backgroundImage = firstImageUrl;

        applyLayout();
        requestAnimationFrame(function () {
            applyLayout();
            track.classList.add('is-ready');
        });
    }

    if (document.readyState === 'complete') {
        initialize();
    } else {
        window.addEventListener('load', initialize);
    }
})();

        // แปลงเลขให้เป็น string สวยๆ ภาษาไทย
        function toThNumber(val) {
            const n = Number(val);
            return Number.isFinite(n) ? n.toLocaleString('th-TH') : String(val ?? '');
        }

        // ล้างค่าที่ไม่ควรแสดง เช่น NULL / undefined / N/A / -
        function cleanVal(v) {
            const s = (v ?? '').toString().trim();
            if (!s) return '';
            return (/^(null|undefined|n\/?a|none|-|\(null\))$/i.test(s)) ? '' : s;
        }

        // สร้าง description จากข้อมูลบนปุ่ม
        function buildDescriptionFromBtn($btn) {
            const getData = (key, attrName) =>
                $btn.data(key) ?? $btn.attr(attrName) ?? '';

            let unitNo = cleanVal(getData('source-id', 'data-source-id'));
            let direction = cleanVal(getData('direction', 'data-direction'));
            let keysale = cleanVal(getData('keysale', 'data-keysale'));
            let dealPrice = cleanVal(getData('deal-price', 'data-deal-price'));
            let promo = cleanVal(getData('promotion', 'data-promotion'));
            let promoOn = cleanVal(getData('promotion-ontop', 'data-promotion-ontop'));
            let flashOn = cleanVal(getData('flashdeal-ontop', 'data-flashdeal-ontop'));

            const $card = $btn.closest('.card-unit');
            if (!direction || !keysale) {
                const $spans = $card.find('.text-detail-unit span');
                direction = direction || cleanVal(($spans.eq(-2).text() || ''));
                keysale = keysale || cleanVal(($spans.eq(-1).text() || ''));
            }
            if (!dealPrice) {
                const raw = ($card.find('.h-deal-price').text() || '').replace(/[^\d.,-]/g, '');
                dealPrice = cleanVal(raw);
            }

            // รวมโปรหลัก
            const promoParts = [promo, promoOn].filter(Boolean);
            const promoCombo = promoParts.join(' + ');

            // จัด TOP UP เมื่อ (1) มีค่า flashOn และ (2) แคมเปญยัง active
            const flashTxt = flashOn ? toThNumber(flashOn) : '';

                : '';

            const desc =
                `Website_Unit : ${unitNo} ${direction} ${keysale} พิเศษ ${dealPrice} ล้านบาท` +
                (promoCombo ? ` ${promoCombo}` : '') +
                flashSegment;

        }

    (function () {
    var MAX_VISIBLE = 3;
    var TEXT_MORE = 'ดูทั้งหมด';
    var TEXT_LESS = 'แสดงน้อยลง';
    var initialized = false;

    function getCollapsedHeight(list) {
        var items = list.querySelectorAll('.list-distance-unit');
        var height = 0;
        for (var j = 0; j < MAX_VISIBLE && j < items.length; j++) {
            height += items[j].offsetHeight + 10;
        }
        return height;
    }

    function handleResize() {
        var isMobile = window.innerWidth <= 991;
        var lists = document.querySelectorAll('.poi-route-list');

        lists.forEach(function (list) {
            var items = list.querySelectorAll('.list-distance-unit');
            if (items.length <= MAX_VISIBLE) return;

            var btn = list.parentElement.querySelector('.btn-show-more-poi');
            if (!btn) return;

            if (isMobile) {
                btn.classList.add('active');
                if (list.getAttribute('data-expanded') !== 'true') {
                    list.style.maxHeight = getCollapsedHeight(list) + 'px';
                }
                if (!initialized) {
                    btn.addEventListener('click', function () {
                        var btnText = btn.querySelector('.btn-poi-text');
                        var isCollapsed = list.getAttribute('data-expanded') !== 'true';
                        if (isCollapsed) {
                            list.style.maxHeight = list.scrollHeight + 'px';
                            list.setAttribute('data-expanded', 'true');
                            btn.classList.add('expanded');
                            if (btnText) btnText.textContent = TEXT_LESS;
                        } else {
                            list.style.maxHeight = getCollapsedHeight(list) + 'px';
                            list.setAttribute('data-expanded', 'false');
                            btn.classList.remove('expanded');
                            if (btnText) btnText.textContent = TEXT_MORE;
                        }
                    });
                }
            } else {
                // Desktop: reset to full height
                list.style.maxHeight = '';
                list.removeAttribute('data-expanded');
                btn.classList.remove('active');
                btn.classList.remove('expanded');
                var btnText = btn.querySelector('.btn-poi-text');
                if (btnText) btnText.textContent = TEXT_MORE;
            }
        });

        initialized = true;
    }

    var resizeTimer;
    window.addEventListener('resize', function () {
        clearTimeout(resizeTimer);
        resizeTimer = setTimeout(handleResize, 150);
    });

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', handleResize);
    } else {
        handleResize();
    }
})();

(function () {
    var stickyBar = document.getElementById('stickyBar');
    var heroBanner = document.getElementById('HeroBanner');
    var boxContact = document.getElementById('boxContact');

    var visibleSections = new Set();

    var observer = new IntersectionObserver(function (entries) {
        entries.forEach(function (entry) {
            if (entry.isIntersecting) {
                visibleSections.add(entry.target);
            } else {
                visibleSections.delete(entry.target);
            }
        });

        if (visibleSections.size > 0) {
            stickyBar.classList.add('sticky-hidden');
        } else {
            stickyBar.classList.remove('sticky-hidden');
            stickyBar.style.animation = 'none';
            stickyBar.offsetHeight;
            stickyBar.style.animation = '';
        }
    }, { threshold: 0 });

    if (stickyBar && heroBanner) observer.observe(heroBanner);
    if (stickyBar && boxContact) observer.observe(boxContact);
})();

        (function() {
            var stickyVdoLoaded = false;
            function loadStickyVdo() {
                if (stickyVdoLoaded) return;
                if (window.scrollY < 200 && document.documentElement.scrollTop < 0) return;
                stickyVdoLoaded = true;
                window.removeEventListener('scroll', loadStickyVdo);
                fetch('https://www.sansiri.com/services/api/get-social-vdo-sticky?projectId=529')
                .then(function(res) { return res.text(); })
                .then(function(html) {
                    if (html && html.trim() !== '') {
                        var container = document.createElement('div');
                        container.innerHTML = html;
                        var children = container.childNodes;
                        while (children.length > 0) {
                            var node = children[0];
                            if (node.nodeName === 'SCRIPT') {
                                var script = document.createElement('script');
                                script.textContent = node.textContent;
                                document.body.appendChild(script);
                                container.removeChild(node);
                            } else {
                                document.body.appendChild(node);
                            }
                        }
                    }
                })
                .catch(function(err) {
                    console.error('SocialVdoSticky error:', err);
                });
            }
            window.addEventListener('scroll', loadStickyVdo, { passive: true });
        })();
    
    
        (function () {
            function isInViewport(el) {
                if (!el) return false;
                var rect = el.getBoundingClientRect();
                return rect.top < window.innerHeight && rect.bottom > 0;
            }
    
            function checkVisibility() {
                var stickyVdo = document.getElementById('stickyVdoWrapper');
                if (!stickyVdo) return;
                var heroBanner = document.getElementById('boxVisual');
                var boxContact = document.getElementById('boxContact');
                var shouldHide = isInViewport(heroBanner) || isInViewport(boxContact);
    
                if (shouldHide) {
                    stickyVdo.classList.remove('is-visible');
                } else {
                    stickyVdo.classList.add('is-visible');
                }
            }
    
            window.addEventListener('scroll', checkVisibility);
            checkVisibility();
        })();
